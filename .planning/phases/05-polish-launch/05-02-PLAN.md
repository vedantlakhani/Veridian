import {
  useSharedValue,
  interpolate,
  Extrapolation,              // NOT Extrapolate — this is Reanimated 4.x
  useAnimatedScrollHandler,
  useAnimatedStyle,
  Animated
} from 'react-native-reanimated';

const SCREEN_WIDTH = Dimensions.get('window').width;
const scrollX = useSharedValue(0);
const scrollHandler = useAnimatedScrollHandler({
  onScroll: (event) => { scrollX.value = event.contentOffset.x; },
});

// Dot indicator style per slide index
const dotStyle = useAnimatedStyle(() => ({
  opacity: interpolate(
    scrollX.value,
    [(index - 1) * SCREEN_WIDTH, index * SCREEN_WIDTH, (index + 1) * SCREEN_WIDTH],
    [0.3, 1, 0.3],
    Extrapolation.CLAMP   // <-- Extrapolation (not Extrapolate)
  ),
  width: interpolate(
    scrollX.value,
    [(index - 1) * SCREEN_WIDTH, index * SCREEN_WIDTH, (index + 1) * SCREEN_WIDTH],
    [8, 24, 8],
    Extrapolation.CLAMP
  ),
}));

// FlatList must be Animated.FlatList
<Animated.FlatList
  data={SLIDES}
  horizontal
  pagingEnabled
  showsHorizontalScrollIndicator={false}
  onScroll={scrollHandler}
  scrollEventThrottle={16}
  renderItem={({ item, index }) => <SlideScreen slide={item} scrollX={scrollX} index={index} />}
  keyExtractor={(_, i) => String(i)}
/>
```

import { create } from 'zustand';
import AsyncStorage from '@react-native-async-storage/async-storage';

const ONBOARDING_KEY = '@veridian/onboarding_complete';

interface OnboardingState {
  onboardingComplete: boolean;
  isChecked: boolean;
  initialize: () => Promise<void>;
  complete: () => Promise<void>;
}

export const useOnboardingStore = create<OnboardingState>((set) => ({
  onboardingComplete: false,
  isChecked: false,
  initialize: async () => {
    const value = await AsyncStorage.getItem(ONBOARDING_KEY);
    set({ onboardingComplete: value === 'true', isChecked: true });
  },
  complete: async () => {
    await AsyncStorage.setItem(ONBOARDING_KEY, 'true');
    set({ onboardingComplete: true });
  },
}));
```

TypeScript strict: no `any`, no `@ts-ignore`. Export only `useOnboardingStore`.
