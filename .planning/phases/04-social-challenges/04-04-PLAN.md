    useAchievements(userId: string | undefined):
    - queryKey: ['achievements', userId]
    - queryFn: fetch all achievements (supabase.from('achievements').select('*')) and user_achievements with joined achievement details (supabase.from('user_achievements').select('*, achievements(*)').eq('user_id', userId))
    - Returns { achievements: Achievement[]; earned: UserAchievementWithDetails[] }
    - enabled: !!userId

    checkAndUnlockAchievements(userId: string): Promise<Achievement[]>
    - NOT a hook — plain async function (called from mutation onSuccess)
    - Fetches all achievements and earned achievement IDs
    - Filters unearned achievements
    - For each unearned achievement, calls checkCriteria(achievement, userId)
    - If met: INSERT into user_achievements { user_id: userId, achievement_id: achievement.id }
    - Returns array of newly earned Achievement objects (caller shows toast for each)

    checkCriteria(achievement: Achievement, userId: string): Promise<boolean>
    - NOT exported — internal helper
    - Switch on achievement.criteria_type:
      - 'first_log': const { count } = await supabase.from('emission_entries').select('*', { count: 'exact', head: true }).eq('user_id', userId); return (count ?? 0) > 0
      - 'streak_days': query daily_summaries WHERE user_id = userId ORDER BY date DESC; compute consecutive days count from today; return count >= achievement.criteria_value
        - Streak logic: const today = getLocalDateString(); fetch rows ordered by date DESC; walk forward in sorted dates; count consecutive days where date[i] = date[i-1] minus 1 day
      - 'reduction_pct': query weekly_summaries WHERE user_id = userId ORDER BY week_start DESC LIMIT 2; if < 2 rows return false; reduction = (rows[1].total_kg_co2e - rows[0].total_kg_co2e) / rows[1].total_kg_co2e * 100; return reduction >= achievement.criteria_value
        - Note: rows[0] = current week (most recent), rows[1] = prior week
      - 'total_entries': const { count } = await supabase.from('emission_entries').select('*', { count: 'exact', head: true }).eq('user_id', userId); return (count ?? 0) >= achievement.criteria_value
      - default: return false

    Import getLocalDateString from '@/lib/emissions' for streak date comparisons.
  </behavior>
